% Plantilla del libro. `herramientas/a_latex.py` la lee y sustituye los marcadores {{...}}.
%
% Está separada del script a propósito: el diseño de un libro se toca muchas más veces que el
% conversor, y quien quiera cambiar tipografía, caja o portada no debería tener que leer Python.
% La sustitución es un `str.replace` literal de cada marcador, NO un `str.format`, porque LaTeX
% está lleno de llaves y `format` se ahogaría en la primera macro.
%
% Marcadores disponibles (se escriben con dobles llaves; aquí van con paréntesis a propósito,
% porque un marcador de verdad dentro de este comentario también se sustituiría, y el cuerpo, que
% ocupa muchas líneas, se saldría del comentario y rompería la compilación):
%   ((TITULO))      título de la novela, ya escapado
%   ((LOGLINE))     la logline de premisa.md, ya escapada (vacía si no había)
%   ((PIE_PORTADA)) línea de procedencia (nombre de la carpeta de la novela)
%   ((CUERPO))      los capítulos generados, cada uno con \capitulo{num}{título}

% `book` y no `article`: queremos páginas impares para los capítulos, portada aparte y un libro a
% doble cara aunque se lea en pantalla. `openright` fuerza que cada capítulo caiga en impar.
\documentclass[11pt,twoside,openright]{book}

% El texto viene en UTF-8 con acentos, eñes y comillas angulares. Con pdfTeX hay que declararlo;
% con XeTeX/LuaTeX el UTF-8 es nativo y en su lugar se usa fontspec.
\usepackage{iftex}
\ifPDFTeX
  \usepackage[T1]{fontenc}
  \usepackage[utf8]{inputenc}
\else
  \usepackage{fontspec}
  \defaultfontfeatures{Ligatures=TeX}
\fi

% Tipografía: TeX Gyre Pagella (un Palatino libre). Se elige por tres razones concretas:
% ojo grande y legible en cuerpo 11, buen color de página en párrafos largos, y —lo que más pesa
% aquí— acentos, eñes, «guillemets» y raya de diálogo completos en la misma fuente, sin recurrir a
% glifos prestados de otra familia. Además va en cualquier distribución TeX moderna.
% Si no estuviera, el documento sigue compilando con Latin Modern en vez de romperse.
\ifPDFTeX
  \IfFileExists{tgpagella.sty}{\usepackage{tgpagella}}{\usepackage{lmodern}}
\else
  \IfFontExistsTF{TeX Gyre Pagella}{\setmainfont{TeX Gyre Pagella}}{}
\fi

% Español: partición de palabras, «Capítulo» y «Índice» en castellano.
% `es-noshorthands` es obligatorio aquí y no es cosmético: babel-spanish convierte por defecto
% `"`, `<`, `>` y `~` en atajos activos, y este texto lo escribe un modelo. Un `"` suelto en la
% prosa haría saltar el compilador o cambiaría el glifo sin avisar. Mejor sin atajos.
\usepackage[spanish,es-noshorthands]{babel}

% Tamaño de libro, no A4: 13,5 x 21 cm es el formato de bolsillo/trade de novela en castellano.
% La caja resultante (unos 9,8 cm) deja renglones de 60-65 caracteres, que es el tramo donde el ojo
% no se pierde al volver al margen. Margen interior mayor que el exterior porque el cosido se come
% parte del blanco del lomo.
\usepackage[
  paperwidth=13.5cm, paperheight=21cm,
  inner=2.1cm, outer=1.7cm,
  top=2.0cm, bottom=2.3cm,
  footskip=1.1cm, headsep=0.7cm
]{geometry}

% microtype afina el gris de la página (protrusión y expansión). Degrada solo si el motor no la
% soporta del todo, así que se carga sin condiciones más allá de que exista.
\IfFileExists{microtype.sty}{\usepackage{microtype}}{}

% Interlineado algo abierto: con Pagella en 11pt el leading por defecto aprieta demasiado para
% leer seguido durante páginas.
\linespread{1.10}

% Sangría de primera línea en vez de espacio entre párrafos: es lo que hace una novela, y lo que
% distingue visualmente un párrafo de diálogo nuevo del anterior.
\setlength{\parindent}{1.2em}
\setlength{\parskip}{0pt}

% Sin numeración de secciones tipo «1.1»: en una novela no hay secciones que numerar.
\setcounter{secnumdepth}{-1}
\setcounter{tocdepth}{0}

% Evitar líneas viudas y huérfanas cuesta poco y se nota en cada página.
\widowpenalty=10000
\clubpenalty=10000

% Cabeceras: título de la novela en la par, título del capítulo en la impar. fancyhdr no está en
% todas las instalaciones mínimas, así que si falta se cae a numeración simple al pie.
\IfFileExists{fancyhdr.sty}{%
  \usepackage{fancyhdr}
  \pagestyle{fancy}
  \fancyhf{}
  \renewcommand{\headrulewidth}{0pt}
  \fancyhead[LE]{\footnotesize\scshape {{TITULO}}}
  \fancyhead[RO]{\footnotesize\scshape\nouppercase{\leftmark}}
  \fancyfoot[C]{\footnotesize\thepage}
  % La primera página de cada capítulo va limpia: sin cabecera, solo el folio.
  \fancypagestyle{plain}{\fancyhf{}\renewcommand{\headrulewidth}{0pt}\fancyfoot[C]{\footnotesize\thepage}}
}{\pagestyle{plain}}

% ---------------------------------------------------------------------------
% Macros que usa el cuerpo generado. Tocar aquí cambia el diseño sin tocar Python.
% ---------------------------------------------------------------------------

% Apertura de capítulo hecha a mano en lugar de con titlesec: son doce líneas, se entienden de un
% vistazo y no añaden una dependencia más que podría faltar. `\cleardoublepage` es lo que garantiza
% que cada capítulo empiece en página nueva (y en impar, por `openright`).
% #1 = número de capítulo, #2 = título de la escaleta.
\newcommand{\capitulo}[2]{%
  \cleardoublepage
  \thispagestyle{plain}%
  \markboth{#2}{#2}%
  % Como no usamos \chapter, el índice hay que alimentarlo a mano.
  \addcontentsline{toc}{chapter}{#1.~#2}%
  \vspace*{0.10\textheight}%
  {\centering
    {\footnotesize\scshape\rotulodecapitulo~#1}\par
    \vspace{0.9em}
    {\Large #2\par}
    \vspace{0.6em}
    \rule{3.2em}{0.4pt}\par
  }
  \vspace{2.2em}%
  % El primer párrafo tras un título va sin sangrar, como en imprenta.
  \noindent\ignorespaces
}
% babel define \chaptername = «Capítulo»; se guarda en un alias por si alguien cambia de idioma.
% El nombre es largo y feo a propósito: se llamaba \spanishchaptername y babel-spanish YA define esa
% macro, así que la plantilla la pisaba y LaTeX abortaba con «already defined». No se vio hasta la
% primera compilación de verdad, que es exactamente la clase de fallo que la validación estructural
% del conversor no puede encontrar: está en la plantilla, no en el texto.
\newcommand{\rotulodecapitulo}{\chaptername}

% Corte de escena dentro de un capítulo: el clásico asterisco centrado.
\newcommand{\escena}{%
  \par\vspace{1.1em}{\centering *\par}\vspace{1.1em}\noindent\ignorespaces
}

% Párrafo de diálogo. Hoy no hace nada distinto —en español el diálogo con raya se sangra igual que
% la prosa— pero existe para que el conversor lo marque y quien rediseñe el libro pueda cambiar
% todos los diálogos de golpe desde aquí.
\newcommand{\dialogo}[1]{#1}

% La raya española (—) abre el parlamento pegada a la palabra y no admite que se quede sola al
% final del renglón. Esto lo asegura: si el conversor emite \raya, nunca habrá un salto detrás.
\newcommand{\raya}{\textemdash\nobreak}
